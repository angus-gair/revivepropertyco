function webhookMiddleware(secret, maxAgeSeconds = 300) {
  return (req, res, next) => {
    const timestamp = req.headers['x-twenty-timestamp'] || req.headers['twenty-timestamp'];
    const signature = req.headers['x-twenty-signal'] || req.headers['twenty-signature'];

    if (!timestamp || !signature) {
      return res.status(401).json({
        success: false,
        error: 'Missing webhook headers'
      });
    }

    const rawPayload = req.body.toString();
    if (verifyWebhookSignature(rawPayload, timestamp, signature, secret, maxAgeSeconds)) {
      req.webhookPayload = JSON.parse(rawPayload);
      next();
    } else {
      return res.status(401).json({ error: 'Invalid signature' });
    }
  };
}

        const duplicate = await isDuplicateEvent(eventId);

| T-02-32 | Replay Attack | Old webhooks | mitigate | Timestamp validation rejects events older than 5 minutes |
